    std::string Operation::formatErr(std::string text)
    {
        std::string phaseErr;

        switch (phase)
        {
        case OpState::Created:
            phaseErr = "InitError";
            break;
        case OpState::Initialing:
            phaseErr = "InitError";
            break;
        case OpState::Ready:
            phaseErr = "InitError";
            break;
        case OpState::Compile:
            phaseErr = "CompileError";
            break;
        case OpState::Running:
            phaseErr = "RuntimeError";
            break;
        case OpState::Stopped:
            phaseErr = "UnknownError";
            break;
        default:
            phaseErr = "UnknownError";
            break;
        }
        std::string message = "[" + phaseErr + "]:" + text;
        return message;
    }
